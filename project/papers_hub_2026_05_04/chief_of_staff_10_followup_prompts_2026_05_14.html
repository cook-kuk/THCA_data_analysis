def fetch_pubmed(query, since_days=30, retmax=25):
    h = Entrez.esearch(db="pubmed", term=query, datetype="pdat", reldate=since_days, retmax=retmax)
    ids = Entrez.read(h)["IdList"]
    if not ids: return []
    h = Entrez.efetch(db="pubmed", id=",".join(ids), rettype="abstract", retmode="xml")
    return Entrez.read(h).get("PubmedArticle", [])

# Cached system prompt — keep stable across daily/weekly runs to maximize cache hits

    {
        "type": "text",
        "text": (
            "You summarize biomedical literature for a thyroid cancer dark-matter "
            "research program. Be concise. Tag each item with relevance "
            "(P1=core, P2=adjacent, P3=context) and flag any paper that would block "
            "or unblock Paper 1 (DM1), Paper 2 (image-DM1), HLA #1 (HT-overlap PTC), "
            "or Paper 4 (Korean GD/Pan-Asian). NEVER fabricate citations."
        ),
        "cache_control": {"type": "ephemeral"},
    }
]

        f"PMID {p['MedlineCitation']['PMID']}: "

    ) or "(no new papers)"
    msg = client.messages.create(
        model="claude-opus-4-7",
        max_tokens=1500,
        system=SYSTEM,
        messages=[{"role": "user", "content": f"Query: {query}\nPapers:\n{payload}"}],
    )
    return msg.content[0].text

def main():
    today = datetime.date.today().isoformat()
    out = {"date": today, "queries": []}
    for q in QUERIES:
        papers = fetch_pubmed(q)
        digest = summarize(papers, q)
        out["queries"].append({"query": q, "n": len(papers), "digest": digest})
    path = f"project/papers_hub_2026_05_04/literature_watch_{today}.json"
    with open(path, "w") as f:
        json.dump(out, f, indent=2)
    print(f"wrote {path}")

if __name__ == "__main__":
    main()
